cmake_minimum_required(VERSION 3.22)
project(novapinyin VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(GNUInstallDirs)
include(CTest)
find_package(Fcitx5Core 5.0.21 REQUIRED)
find_package(Fcitx5Config REQUIRED)
find_package(LibIMEPinyin 1.1.5 REQUIRED)
find_file(NOVA_MODEL_FILE zh_CN.lm PATHS /usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}/libime /usr/share/libime REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENCC REQUIRED IMPORTED_TARGET opencc)
add_library(nova-core STATIC src/core/core.cpp src/core/store.cpp src/core/context.cpp src/core/completion.cpp)
target_include_directories(nova-core PUBLIC src)
target_link_libraries(nova-core PUBLIC LibIME::Pinyin SQLite::SQLite3 Threads::Threads PkgConfig::OPENCC)
target_compile_options(nova-core PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(nova-core PUBLIC NOVA_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/novapinyin")
target_compile_definitions(nova-core PRIVATE NOVA_MODEL_FILE="${NOVA_MODEL_FILE}")
add_library(novapinyin MODULE src/addon/addon.cpp)
target_link_libraries(novapinyin PRIVATE nova-core Fcitx5::Core Fcitx5::Config)
target_compile_options(novapinyin PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS novapinyin DESTINATION ${CMAKE_INSTALL_LIBDIR}/fcitx5)
install(FILES data/novapinyin-addon.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/addon RENAME novapinyin.conf)
install(FILES data/novapinyin.conf DESTINATION ${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod)
install(FILES data/semiconductor.tsv data/programming.tsv DESTINATION ${CMAKE_INSTALL_DATADIR}/novapinyin)
add_executable(novapinyin-tool tools/tool.cpp)
target_link_libraries(novapinyin-tool PRIVATE nova-core)
install(TARGETS novapinyin-tool RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS tools/novapinyin-manager.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME novapinyin-manager)
install(PROGRAMS tools/novapinyin-project.py DESTINATION ${CMAKE_INSTALL_BINDIR} RENAME novapinyin-project)
install(FILES data/novapinyin-manager.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES README.md docs/install.md docs/dictionary-format.md docs/validation.md docs/advanced-input.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/novapinyin)
if(BUILD_TESTING)
  find_package(GTest REQUIRED)
  add_executable(nova-tests tests/core_test.cpp tests/advanced_test.cpp)
  target_link_libraries(nova-tests PRIVATE nova-core GTest::gtest_main)
  target_compile_definitions(nova-tests PRIVATE NOVA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  include(GoogleTest)
  gtest_discover_tests(nova-tests)
  add_executable(addon-tests tests/addon_test.cpp src/addon/addon.cpp)
  target_link_libraries(addon-tests PRIVATE nova-core Fcitx5::Core Fcitx5::Config GTest::gtest_main)
  add_test(NAME addon-events COMMAND addon-tests)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME project-index COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/project_test.py)
endif()
if(BUILD_TESTING)
  add_executable(nova-benchmark tests/benchmark.cpp)
  target_link_libraries(nova-benchmark PRIVATE nova-core)
endif()
option(NOVA_BACKEND_PROBE "Build optional libpinyin technical probe" OFF)
if(NOVA_BACKEND_PROBE)
  pkg_check_modules(LIBPINYIN REQUIRED IMPORTED_TARGET libpinyin)
  add_executable(probe-libpinyin tools/probe-libpinyin.cpp)
  target_link_libraries(probe-libpinyin PRIVATE PkgConfig::LIBPINYIN)
endif()
option(NOVA_GUI_SMOKE "Build real GTK3/Qt5 input clients for isolated graphical smoke tests" OFF)
if(NOVA_GUI_SMOKE)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
  find_package(Qt5 REQUIRED COMPONENTS Widgets)
  add_executable(nova-gtk-smoke tests/gtk_smoke.cpp)
  target_link_libraries(nova-gtk-smoke PRIVATE PkgConfig::GTK3)
  add_executable(nova-qt-smoke tests/qt_smoke.cpp)
  target_link_libraries(nova-qt-smoke PRIVATE Qt5::Widgets)
endif()
